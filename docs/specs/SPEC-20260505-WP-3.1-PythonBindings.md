## Invarianten

1. **Keine GIL-Locks in I/O-Paths** — PyO3 `py.allow_threads()` überall
2. **Numpy-Arrays zero-copy** (kein Kopieren von Vektordaten Python→Rust)
3. **Async-transparent**: Python-seitig sync, Rust-seitig tokio (interner Runtime)
4. **Keine Breaking-Changes** an der Rust-API durch diese Bindings

## Acceptance Criteria (Triple-Test)

```python
# tests/test_bindings.py
import memfuse, numpy as np

def test_open_and_basic_insert_search():
    db = memfuse.open("/tmp/test_wp31", dimension=4)
    col = db.collection("test", dimension=4)
    v = np.array([0.1, 0.2, 0.3, 0.4], dtype=np.float32)
    col.insert("doc1", v)
    results = col.search(v, k=1)
    assert results[0]["id"] == "doc1"  # AC-1

def test_hybrid_search():
    db = memfuse.open("/tmp/test_wp31_hybrid", dimension=4)
    col = db.collection("docs", dimension=4)
    v = np.array([0.1, 0.2, 0.3, 0.4], dtype=np.float32)
    col.insert("doc1", v, metadata={"text": "rust programming language"})
    results = col.hybrid_search("rust programming", v, k=1)
    assert results[0]["id"] == "doc1"  # AC-2

def test_collection_isolation():
    db = memfuse.open("/tmp/test_wp31_iso", dimension=4)
    col_a = db.collection("a", dimension=4)
    col_b = db.collection("b", dimension=4)
    v = np.ones(4, dtype=np.float32)
    col_a.insert("k1", v)
    assert len(col_b.search(v, k=5)) == 0  # AC-3
```
